# Callaway & Sant'Anna (2021) — Staggered DiD
library(did)
data(mpdta)  # Minimum Wage Panel Data · 2,500 obs

out <- att_gt(
  yname = "lemp",  gname = "first.treat",
  idname = "countyreal",  tname = "year",
  data = mpdta,  control_group = "nevertreated",
  est_method = "dr",  print_details = FALSE
)
agg <- aggte(out, type = "simple")
summary(agg)
